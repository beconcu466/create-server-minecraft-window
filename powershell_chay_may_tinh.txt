function Set-MinecraftProperty {
    param (
        [string[]]$Lines,
        [string]$Key,
        [string]$Value
    )

    $Found = $false
    $NewLines = @()

    foreach ($Line in $Lines) {

        if ($Line -match "^$([regex]::Escape($Key))=") {

            if (-not $Found) {
                $NewLines += "$Key=$Value"
                $Found = $true
            }

        }
        else {
            $NewLines += $Line
        }
    }

    if (-not $Found) {
        $NewLines += "$Key=$Value"
    }

    return $NewLines
}

# 6. Firewall
# ============================================================

Write-Host ""
Write-Host "[CHECK] Windows Firewall..." -ForegroundColor Yellow

$FirewallRuleName = "Minecraft Java Server TCP 25565"

try {

    $FirewallRule = Get-NetFirewallRule `
        -DisplayName $FirewallRuleName `
        -ErrorAction SilentlyContinue

    if ($FirewallRule) {

        Write-Host "[SKIP] Firewall rule đã tồn tại." -ForegroundColor Green

    }
    else {

        New-NetFirewallRule `
            -DisplayName $FirewallRuleName `
            -Direction Inbound `
            -Protocol TCP `
            -LocalPort 25565 `
            -Action Allow `
            -Profile Private `
            -ErrorAction Stop | Out-Null

        Write-Host "[OK] Đã mở port 25565." -ForegroundColor Green
    }

}
catch {

    Write-Host "[WARNING] Không thể tự mở Firewall." -ForegroundColor Yellow
    Write-Host "Nếu máy khác không vào được, hãy chạy PowerShell bằng Administrator."
}
